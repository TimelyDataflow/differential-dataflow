# Shortest paths with delta stepping: see shortest_paths.ddp.
# Run from interactive/: ddir_server < examples/recipes/shortest_paths.txt
load sp from examples/recipes/shortest_paths.ddp

# Time 0: 1 -> 2 -> 3 costs 200 + 200; the direct edge 1 -> 3 costs 500.
feed sp 0 1 val=tuple(2,200) time=0
feed sp 0 2 val=tuple(3,200) time=0
feed sp 0 1 val=tuple(3,500) time=0
feed sp 0 3 val=tuple(4,10) time=0
feed sp 1 1 time=0
seal sp 0 1
seal sp 1 1
peek sp::dist

# Time 1: the edge 2 -> 3 gets dearer, and the direct edge wins.
feed sp 0 2 val=tuple(3,200) time=1 diff=-1
feed sp 0 2 val=tuple(3,900) time=1
seal sp 0 2
seal sp 1 2
peek sp::dist
exit
