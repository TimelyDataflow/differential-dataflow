# Top k through a tree of `reduce(limit k)`: see top_k.ddp.
# Run from interactive/: ddir_server < examples/recipes/top_k.txt
load top from examples/recipes/top_k.ddp

# Time 0: five players.
feed top 0 1 val=50 time=0
feed top 0 2 val=80 time=0
feed top 0 65 val=70 time=0
feed top 0 3 val=10 time=0
feed top 0 129 val=90 time=0
seal top 0 1
peek top::top

# Time 1: player 3 climbs to the top, and player 129 drops out: a score change is a retraction
# of the old row and an insertion of the new one.
feed top 0 3 val=10 time=1 diff=-1
feed top 0 3 val=95 time=1
feed top 0 129 val=90 time=1 diff=-1
feed top 0 129 val=20 time=1
seal top 0 2
peek top::top
exit
