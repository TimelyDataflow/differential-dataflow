# Triangles as a delta join: see triangles.ddp.
# Run from interactive/: ddir_server < examples/recipes/triangles.txt
load tri from examples/recipes/triangles.ddp

# Time 0: all six edges among 1, 2, 3, 4 (four triangles), and the edge 4-5 (none).
feed tri 0 1 val=2 time=0
feed tri 0 1 val=3 time=0
feed tri 0 1 val=4 time=0
feed tri 0 2 val=3 time=0
feed tri 0 2 val=4 time=0
feed tri 0 3 val=4 time=0
feed tri 0 4 val=5 time=0
seal tri 0 1
peek tri::total
peek tri::triangles

# Time 1: removing 1-2 leaves the two triangles through 3-4.
feed tri 0 1 val=2 time=1 diff=-1
seal tri 0 2
peek tri::total
peek tri::triangles
exit
