# Reading results back out with `peek`.
load prod from examples/server/programs/producer.ddp
load reach from examples/server/programs/reach_import.ddp edges=prod::edges

# graph 1 -> 2 -> 3, root 1
feed prod 0 1,2 time=0
feed prod 0 2,3 time=0
feed reach roots 1 time=0
seal prod 0 1
seal reach roots 1

# Snapshot the shared edge trace, and the reachability result.
peek prod::edges
peek reach::result
# Just one key of the result.
peek reach::result 2
# An unknown trace reports an error, not a crash.
peek nope
exit
