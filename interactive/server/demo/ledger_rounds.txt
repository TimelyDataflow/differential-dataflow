# Ledger transactions, resolved in rounds on the stock server (see interactive/model/resolve.ddp and model.md).
#
# `state` holds the manifest and the three indexes the rules read. Each round loads the rules against it,
# feeds the round's requests, ticks, reads the outcomes, feeds the conclusions back, and drops the rules.
# A request is a list of actions: inject(0, (id, shape, width, start, clock)) creates a ledger, and
# inject(1, (id, lower, upper, batch, seqno)) appends a span. Nothing here is durable; for that,
# see interactive/bench/ledger/fabric.py.
#
#   ./target/release/ddir_server < interactive/server/demo/ledger_rounds.txt

load state from interactive/bench/ledger/state.ddp
tick
# Round 1. Ledger ids are (0,n); every time has one coordinate.
#   10 creates clock 8.                         20 creates ledger 3 and appends [0,1) to it.
#   30 creates ledger 1 under clock 8: rejected, since a clock must be registered before the round.
#   40 gives a start of the wrong width.        50 gives an upper of another width than its lower.
#   60 lists its ledgers out of order.          70 appends [1,2) to ledger 3, as version 1.
#   80 appends at the same place: stale, and told ledger 3 now stands at version 2, frontier [2].
#   90 appends to a ledger nobody created.
load res from interactive/model/resolve.ddp input0=state::own input2=state::registered input3=state::follows
feed res 1 begin
10 val=tuple(list(inject(0,tuple(tuple(0,8),list(7),1,list(list(0)),inject(0,tuple())))))
20 val=tuple(list(inject(0,tuple(tuple(0,3),list(7),1,list(list(0)),inject(0,tuple()))),inject(1,tuple(tuple(0,3),list(list(0)),list(list(1)),inject(0,tuple()),0))))
30 val=tuple(list(inject(0,tuple(tuple(0,1),list(7),1,list(list(0)),inject(1,tuple(0,8))))))
40 val=tuple(list(inject(0,tuple(tuple(0,4),list(7),2,list(list(0)),inject(0,tuple())))))
50 val=tuple(list(inject(1,tuple(tuple(0,5),list(list(0)),list(list(1,2)),inject(0,tuple()),0))))
60 val=tuple(list(inject(1,tuple(tuple(0,3),list(list(1)),list(list(2)),inject(0,tuple()),1)),inject(0,tuple(tuple(0,2),list(7),1,list(list(0)),inject(0,tuple())))))
70 val=tuple(list(inject(1,tuple(tuple(0,3),list(list(1)),list(list(2)),inject(0,tuple()),1))))
80 val=tuple(list(inject(1,tuple(tuple(0,3),list(list(1)),list(list(5)),inject(0,tuple()),1))))
90 val=tuple(list(inject(1,tuple(tuple(0,77),list(list(0)),list(list(1)),inject(0,tuple()),0))))
end-feed
tick
peek res::results
peek res::registrations_add
peek res::spans_add
feed state 0 from res::own_delta
feed state 1 from res::registered_add
feed state 2 from res::follows_add
feed state 3 from res::registrations_add
feed state 4 from res::spans_add
drop res
tick
peek state::own
# Round 2.
#   10 creates ledger 1 under clock 8 and appends [0,1).   20 creates a follower of another width than its clock.
#   30 advances clock 8 to [5].                             40 creates clock 9, which follows clock 8.
load res from interactive/model/resolve.ddp input0=state::own input2=state::registered input3=state::follows
feed res 1 begin
10 val=tuple(list(inject(0,tuple(tuple(0,1),list(7),1,list(list(0)),inject(1,tuple(0,8)))),inject(1,tuple(tuple(0,1),list(list(0)),list(list(1)),inject(0,tuple()),0))))
20 val=tuple(list(inject(0,tuple(tuple(0,6),list(7),2,list(list(0,0)),inject(1,tuple(0,8))))))
30 val=tuple(list(inject(1,tuple(tuple(0,8),list(list(0)),list(list(5)),inject(0,tuple()),0))))
40 val=tuple(list(inject(0,tuple(tuple(0,9),list(7),1,list(list(0)),inject(1,tuple(0,8))))))
end-feed
tick
peek res::results
feed state 0 from res::own_delta
feed state 1 from res::registered_add
feed state 2 from res::follows_add
feed state 3 from res::registrations_add
feed state 4 from res::spans_add
drop res
tick
# Round 3. Ledger 1's own upper is [1], but its clock stands at [5], and advances to [7] in this round.
#   10 appends to ledger 1 at [1]: stale, and told it stands at version 2, frontier [7], as the round leaves it.
#   20 creates ledger 2 under clock 9 and appends at [0]: stale, since clock 9 follows clock 8, at [5].
#   30 advances clock 8 to [7].                40 appends [5,6) to ledger 1: the clock's advance lands after it.
#   50 creates ledger 2 and appends [5,6).     60 appends to ledger 3 with times of two coordinates.
#   70 creates ledger 12 and appends at [4]: stale against its own creation, which starts at [0].
load res from interactive/model/resolve.ddp input0=state::own input2=state::registered input3=state::follows
feed res 1 begin
10 val=tuple(list(inject(1,tuple(tuple(0,1),list(list(1)),list(list(2)),inject(0,tuple()),1))))
20 val=tuple(list(inject(0,tuple(tuple(0,2),list(7),1,list(list(0)),inject(1,tuple(0,9)))),inject(1,tuple(tuple(0,2),list(list(0)),list(list(1)),inject(0,tuple()),0))))
30 val=tuple(list(inject(1,tuple(tuple(0,8),list(list(5)),list(list(7)),inject(0,tuple()),1))))
40 val=tuple(list(inject(1,tuple(tuple(0,1),list(list(5)),list(list(6)),inject(0,tuple()),1))))
50 val=tuple(list(inject(0,tuple(tuple(0,2),list(7),1,list(list(0)),inject(1,tuple(0,9)))),inject(1,tuple(tuple(0,2),list(list(5)),list(list(6)),inject(0,tuple()),0))))
60 val=tuple(list(inject(1,tuple(tuple(0,3),list(list(2,0)),list(list(3,0)),inject(0,tuple()),2))))
70 val=tuple(list(inject(0,tuple(tuple(0,12),list(7),1,list(list(0)),inject(0,tuple()))),inject(1,tuple(tuple(0,12),list(list(4)),list(list(5)),inject(0,tuple()),0))))
end-feed
tick
peek res::results
peek res::spans_add
feed state 0 from res::own_delta
feed state 1 from res::registered_add
feed state 2 from res::follows_add
feed state 3 from res::registrations_add
feed state 4 from res::spans_add
drop res
tick
peek state::own
peek state::follows
# The model's own account of the manifest agrees with the index, and finds nothing wrong with it.
load check from interactive/model/ledger.ddp input0=state::registrations input1=state::spans
tick
peek check::own
peek check::write
peek check::early
peek check::skipped
exit
